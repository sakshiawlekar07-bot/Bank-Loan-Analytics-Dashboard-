-- total number of records 
SELECT COUNT(*) 
FROM final_cleaned_data_loan;



-- Shows the total number of loans, total loan amount,
-- average loan amount and total amount funded by the bank.
SELECT
    COUNT(`Account ID`) AS Total_Loans,
    SUM(`Loan Amount`) AS Total_Loan_Amount,
    ROUND(AVG(`Loan Amount`), 2) AS Average_Loan_Amount,
    SUM(`Funded Amount`) AS Total_Funded_Amount
FROM final_cleaned_data_loan;



-- Shows the number of loans under each loan status.
-- Calculates total and average loan amounts for each status.

SELECT
    `Loan Status`,
    COUNT(*) AS Total_Loans,
    SUM(`Loan Amount`) AS Total_Loan_Amount,
    ROUND(AVG(`Loan Amount`), 2) AS Average_Loan_Amount

FROM final_cleaned_data_loan

GROUP BY `Loan Status`
ORDER BY Total_Loans DESC;

-- Overall Default and Delinquency Analysis
-- Shows total loans, defaulted loans and delinquent loans.
-- Calculates the default rate and delinquency rate in percentage.

SELECT
    COUNT(*) AS Total_Loans,

    SUM(CASE
        WHEN `Is Default Loan` = 'YES' THEN 1
        ELSE 0
    END) AS Defaulted_Loans,

    ROUND(
        SUM(CASE
            WHEN `Is Default Loan` = 'YES' THEN 1
            ELSE 0
        END) * 100.0 / COUNT(*), 2
    ) AS Default_Rate_Percentage,

    SUM(CASE
        WHEN `Is Delinquent Loan` = 'YES' THEN 1
        ELSE 0
    END) AS Delinquent_Loans,

    ROUND(
        SUM(CASE
            WHEN `Is Delinquent Loan` = 'YES' THEN 1
            ELSE 0
        END) * 100.0 / COUNT(*), 2
    ) AS Delinquency_Rate_Percentage

FROM final_cleaned_data_loan;





-- Query 4: Default Analysis by Loan Grade
-- Shows the total loans and defaulted loans under each grade.
-- Calculates the default percentage for each loan grade.

SELECT
    `Grade`,
    COUNT(*) AS Total_Loans,

    SUM(
        CASE
            WHEN `Is Default Loan` = 'YES' THEN 1
            ELSE 0
        END
    ) AS Defaulted_Loans,

    ROUND(
        SUM(
            CASE
                WHEN `Is Default Loan` = 'YES' THEN 1
                ELSE 0
            END
        ) * 100.0 / COUNT(*), 2
    ) AS Default_Rate_Percentage

FROM final_cleaned_data_loan

WHERE `Grade` IS NOT NULL
AND `Grade` <> 'Null'

GROUP BY `Grade`
ORDER BY Default_Rate_Percentage DESC;





-- Query 5: Default Analysis by Interest Rate
-- Groups loans into different interest rate categories.
-- Calculates the default rate for each category.

SELECT
    CASE
        WHEN `Int Rate` < 0.10 THEN 'Below 10%'
        WHEN `Int Rate` < 0.15 THEN '10% - 15%'
        WHEN `Int Rate` < 0.20 THEN '15% - 20%'
        ELSE 'Above 20%'
    END AS Interest_Rate_Category,

    COUNT(*) AS Total_Loans,

    SUM(
        CASE
            WHEN `Is Default Loan` = 'YES' THEN 1
            ELSE 0
        END
    ) AS Defaulted_Loans,

    ROUND(
        SUM(
            CASE
                WHEN `Is Default Loan` = 'YES' THEN 1
                ELSE 0
            END
        ) * 100.0 / COUNT(*), 2
    ) AS Default_Rate_Percentage

FROM final_cleaned_data_loan

WHERE `Int Rate` IS NOT NULL

GROUP BY Interest_Rate_Category
ORDER BY Default_Rate_Percentage DESC;


-- Query 6: Loan Repayment and Principal Recovery Analysis
-- Compares the total funded amount with the principal amount recovered.
-- Calculates the principal recovery percentage and recovery gap.

SELECT
    SUM(`Funded Amount`) AS Total_Funded_Amount,

    SUM(`Total Rec Prncp`) AS Total_Principal_Recovered,

    ROUND(
        SUM(`Total Rec Prncp`) * 100.0 /
        NULLIF(SUM(`Funded Amount`), 0), 2
    ) AS Principal_Recovery_Percentage,

    SUM(`Funded Amount`) -
    SUM(`Total Rec Prncp`) AS Principal_Recovery_Gap

FROM final_cleaned_data_loan;


-- Query 7: Recovery Analysis by Loan Status
-- Shows the number of loans and total recovery amount
-- for each loan status.

SELECT
    `Loan Status`,

    COUNT(*) AS Total_Loans,

    SUM(`Recoveries`) AS Total_Recovery_Amount,

    ROUND(AVG(`Recoveries`), 2) AS Average_Recovery_Amount,

    SUM(`Collection Recovery fee`) AS Total_Recovery_Fees

FROM final_cleaned_data_loan

GROUP BY `Loan Status`

ORDER BY Total_Recovery_Amount DESC;


-- Query 8: Top 10 Loans with Highest Recovery Amounts
-- Displays loan details and recovery amounts.
-- Helps identify accounts with the highest recovery collections.

SELECT
    `Account ID`,
    `Client Name`,
    `Loan Status`,
    `Loan Amount`,
    `Recoveries`,
    `Total Pymnt`,
    `Is Default Loan`

FROM final_cleaned_data_loan

WHERE `Recoveries` > 0

ORDER BY `Recoveries` DESC

LIMIT 10;


-- Query 9: Loan Distribution by Customer Age Group
-- Groups customers into different age categories.
-- Shows the number of loans and loan amounts for each group.

SELECT
    CASE
        WHEN `Age _T` BETWEEN 18 AND 25 THEN '18-25'
        WHEN `Age _T` BETWEEN 26 AND 35 THEN '26-35'
        WHEN `Age _T` BETWEEN 36 AND 45 THEN '36-45'
        WHEN `Age _T` BETWEEN 46 AND 55 THEN '46-55'
        WHEN `Age _T` > 55 THEN 'Above 55'
        ELSE 'Unknown'
    END AS Age_Group,

    COUNT(*) AS Total_Loans,

    SUM(`Loan Amount`) AS Total_Loan_Amount,

    ROUND(AVG(`Loan Amount`), 2) AS Average_Loan_Amount

FROM final_cleaned_data_loan

WHERE `Age _T` IS NOT NULL

GROUP BY Age_Group

ORDER BY Total_Loans DESC;

-- Query 12: Loan Distribution by Loan Term
-- Compares loan count, loan amount and default rate for each term.

SELECT
    `Term`,
    COUNT(*) AS Total_Loans,
    SUM(`Loan Amount`) AS Total_Loan_Amount,
    ROUND(AVG(`Loan Amount`), 2) AS Average_Loan_Amount,
    SUM(CASE
        WHEN `Is Default Loan` = 'YES' THEN 1
        ELSE 0
    END) AS Defaulted_Loans,
    ROUND(
        SUM(CASE WHEN `Is Default Loan` = 'YES' THEN 1 ELSE 0 END)
        * 100.0 / COUNT(*), 2
    ) AS Default_Rate_Percentage
FROM final_cleaned_data_loan
WHERE `Term` IS NOT NULL
GROUP BY `Term`
ORDER BY Total_Loans DESC;

    SUM(
        CASE
            WHEN `Is Default Loan` = 'YES' THEN 1
